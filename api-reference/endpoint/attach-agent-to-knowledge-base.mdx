````
### Example
**POST with body:**
```python
import requests

def attach_agent_to_knowledge_base(knowledge_base_id: str, agent_id: int, api_key: str) -> dict:
    url = f"https://api.getbluejay.ai/v1/knowledge-base/{knowledge_base_id}/agents/{agent_id}"

    response = requests.post(url, headers=headers)
    response.raise_for_status()
    return response.json()
````
